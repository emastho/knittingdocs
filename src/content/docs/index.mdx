export const quickStart = `import { createPool, isMain } from "knitting";

export const square = (value: number) => value * value;
export const greet = (name: string) => \`hello \${name}\`;

if (isMain) {
  using pool = createPool({ threads: 2 })({ square, greet });

  const [four, message] = await Promise.all([
    pool.call.square(2),
    pool.call.greet("knitting"),
  ]);

  console.log({ four, message });
}`;

    <div class="kh-section-intro">
      <p class="kh-kicker">The mechanics around the call</p>
      <h2 id="kh-features-title">The boundary is part of the API.</h2>
      <p>
        Knitting handles the mechanics that make workers useful in production:
        moving data, isolating code, scheduling work, and preserving stack traces
        and error causes.
      </p>
    </div>

    <div class="kh-feature-grid kh-grid">
      <article class="kh-feature kh-feature--wide">
        <div class="kh-feature-icon" aria-hidden="true">↔</div>
        <p class="kh-feature-label">Fast by construction</p>
        <h3>Move the work, not all the bytes.</h3>
        <p>
          Pass typed payloads and promises. Move large buffers through zero-copy
          handles. Set timeouts, cancel with abort signals, and keep stack traces
          and error causes intact. Knitting's native work stealing lets
          compatible multi-worker pools claim tasks from a shared submit region
          instead of waiting in private request lanes.
        </p>
        <ul class="kh-chip-list" aria-label="Call and payload features">
          <li>Promise inputs</li>
          <li>SharedArrayBuffer</li>
          <li>ProcessSharedBuffer</li>
          <li>BufferReference</li>
          <li>Task timeouts</li>
          <li>Abort signals</li>
          <li>Typed arrays</li>
          <li>Error cause chains</li>
          <li>Native work stealing</li>
        </ul>

        <p>
          The same task API works with runtime threads or isolated processes.

          Limit cooperative work with timeouts and abort signals, or hard-kill
